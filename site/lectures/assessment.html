Challenge 1: Minimal Diffusion on 2D Point Clouds

Step 1: Noise schedule. Precompute all derived quantities from $\beta$:

import torch import torch.nn as nn import matplotlib.pyplot as plt T = 300 betas = torch.linspace(1e-4, 0.02, T) alphas = 1.0 - betas alpha_bar = torch.cumprod(alphas, dim=0) # (T,) sqrt_alpha_bar = torch.sqrt(alpha_bar) # (T,) sqrt_one_minus_alpha_bar = torch.sqrt(1.0 - alpha_bar) # (T,)

Step 2: MLP denoiser with sinusoidal timestep embedding.

class SinusoidalEmbedding(nn.Module): def __init__(self, dim): super().__init__() self.dim = dim def forward(self, t): half = self.dim // 2 freqs = torch.exp(-torch.log(torch.tensor(10000.0)) * torch.arange(half, device=t.device) / half) args = t[:, None].float() * freqs[None, :] # (B, half) return torch.cat([args.sin(), args.cos()], dim=-1) # (B, dim) class Denoiser(nn.Module): def __init__(self, t_dim=64, hidden=256): super().__init__() self.time_emb = SinusoidalEmbedding(t_dim) self.net = nn.Sequential( nn.Linear(2 + t_dim, hidden), nn.GELU(), nn.Linear(hidden, hidden), nn.GELU(), nn.Linear(hidden, hidden), nn.GELU(), nn.Linear(hidden, 2), ) def forward(self, x, t): t_emb = self.time_emb(t) # (B, t_dim) inp = torch.cat([x, t_emb], -1) # (B, 2 + t_dim) return self.net(inp) # (B, 2)

Step 3: Training loop.

model = Denoiser() optimizer = torch.optim.Adam(model.parameters(), lr=3e-4) data = make_spiral(2000) for step in range(5000): idx = torch.randint(0, len(data), (256,)) x_0 = data[idx] # (B, 2) t = torch.randint(0, T, (x_0.shape[0],)) # (B,) noise = torch.randn_like(x_0) # (B, 2) # Forward process: x_t = sqrt(alpha_bar_t) * x_0 + sqrt(1 - alpha_bar_t) * noise x_t = sqrt_alpha_bar[t, None] * x_0 + sqrt_one_minus_alpha_bar[t, None] * noise noise_pred = model(x_t, t) loss = nn.functional.mse_loss(noise_pred, noise) optimizer.zero_grad() loss.backward() optimizer.step() if step % 1000 == 0: print(f"Step {step}, Loss: {loss.item():.4f}")

Step 4: DDPM sampling loop.

@torch.no_grad() def sample(model, n_samples=1000): x = torch.randn(n_samples, 2) # Start from pure noise for t_idx in reversed(range(T)): t = torch.full((n_samples,), t_idx, dtype=torch.long) noise_pred = model(x, t) # DDPM reverse step alpha_t = alphas[t_idx] alpha_bar_t = alpha_bar[t_idx] beta_t = betas[t_idx] mean = (1.0 / torch.sqrt(alpha_t)) * ( x - (beta_t / torch.sqrt(1.0 - alpha_bar_t)) * noise_pred ) if t_idx > 0: noise = torch.randn_like(x) x = mean + torch.sqrt(beta_t) * noise else: x = mean # No noise at final step return x samples = sample(model)

Step 5: Visualize.

fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(12, 5)) ax1.scatter(data[:, 0], data[:, 1], alpha=0.3, s=5) ax1.set_title("Training data") ax2.scatter(samples[:, 0], samples[:, 1], alpha=0.3, s=5, c="orange") ax2.set_title("Generated samples") plt.show()

Step 1: Timestep sub-selection. Uniformly space $N$ steps across the full schedule of $T$ timesteps:

def get_ddim_timesteps(T, num_steps): # Uniform spacing from T-1 down to 0 step_size = T // num_steps timesteps = list(range(0, T, step_size)) # [0, step, 2*step, ...] return list(reversed(timesteps)) # [T-step, ..., step, 0] descending

Step 2: Full DDIM + CFG implementation.

@torch.no_grad() def ddim_sample_cfg(model, schedule, shape, class_label, guidance_scale=4.0, num_steps=50, device="cpu"): alpha_bar = schedule["alphas_cumprod"].to(device) # (T,) T = len(alpha_bar) timesteps = get_ddim_timesteps(T, num_steps) x = torch.randn(shape, device=device) # (B, C, H, W) B = shape[0] for i in range(len(timesteps)): t_cur = timesteps[i] t_next = timesteps[i + 1] if i + 1 < len(timesteps) else 0 t_batch = torch.full((B,), t_cur, device=device, dtype=torch.long) # ── CFG: batched conditional + unconditional ── x_double = torch.cat([x, x], dim=0) # (2B, C, H, W) t_double = torch.cat([t_batch, t_batch], dim=0) # (2B,) labels = torch.cat([ torch.full((B,), class_label, device=device, dtype=torch.long), torch.full((B,), NULL_CLASS, device=device, dtype=torch.long), ], dim=0) # (2B,) noise_pred_double = model(x_double, t_double, labels) eps_cond, eps_uncond = noise_pred_double.chunk(2, dim=0) # CFG formula: eps = eps_uncond + s * (eps_cond - eps_uncond) eps = eps_uncond + guidance_scale * (eps_cond - eps_uncond) # ── DDIM update ── alpha_bar_t = alpha_bar[t_cur] alpha_bar_next = alpha_bar[t_next] if t_next > 0 else torch.tensor(1.0) # Predict x_0 from x_t and predicted noise x0_pred = (x - torch.sqrt(1 - alpha_bar_t) * eps) / torch.sqrt(alpha_bar_t) x0_pred = x0_pred.clamp(-1, 1) # Stability clamp # Direction pointing to x_t dir_xt = torch.sqrt(1 - alpha_bar_next) * eps # DDIM step (deterministic, eta=0) x = torch.sqrt(alpha_bar_next) * x0_pred + dir_xt return x.clamp(-1, 1)

Key differences from DDPM:

  • DDIM predicts $x_0$ at each step, then jumps to the next $\bar\alpha$ level. DDPM adds noise at each step.
  • With $\eta = 0$, DDIM is deterministic—same initial noise gives the same output.
  • Timestep sub-selection lets you skip steps (e.g., 50 steps instead of 1000) because you're not constrained to the Markov chain.

Common mistakes: Using $\alpha_t$ instead of $\bar\alpha_t$. Getting the CFG formula wrong ($\epsilon_\text{uncond} + s \cdot \epsilon_\text{cond}$ instead of the extrapolation form). Forgetting to clamp $x_0$ predictions.

Step 1: Schedule setup.

def setup_schedule(T=1000, beta_start=1e-4, beta_end=0.02, device="cpu"): betas = torch.linspace(beta_start, beta_end, T, device=device) betas = betas.clamp(max=0.999) # Clip for numerical stability alphas = 1.0 - betas alpha_bar = torch.cumprod(alphas, dim=0) return { "betas": betas, # (T,) "alphas": alphas, # (T,) "alphas_cumprod": alpha_bar, # (T,) "sqrt_alphas_cumprod": torch.sqrt(alpha_bar), # (T,) "sqrt_one_minus_alphas_cumprod": torch.sqrt(1.0 - alpha_bar), }

Step 2: Training step. The core of DDPM training in one function:

def train_step(model, x_0, optimizer, schedule): """One DDPM training step. Returns scalar loss.""" B = x_0.shape[0] device = x_0.device T = len(schedule["betas"]) # 1. Sample random timesteps t = torch.randint(0, T, (B,), device=device) # 2. Sample noise noise = torch.randn_like(x_0) # (B, C, H, W) # 3. Forward process — noise the images sqrt_ab = schedule["sqrt_alphas_cumprod"][t] # (B,) sqrt_1_ab = schedule["sqrt_one_minus_alphas_cumprod"][t] # (B,) # Reshape for broadcasting: (B,) -> (B, 1, 1, 1) for image tensors sqrt_ab = sqrt_ab[:, None, None, None] sqrt_1_ab = sqrt_1_ab[:, None, None, None] x_t = sqrt_ab * x_0 + sqrt_1_ab * noise # (B, C, H, W) # 4. Predict noise noise_pred = model(x_t, t) # (B, C, H, W) # 5. Simple MSE loss on noise loss = nn.functional.mse_loss(noise_pred, noise) # 6. Gradient step optimizer.zero_grad() loss.backward() torch.nn.utils.clip_grad_norm_(model.parameters(), 1.0) optimizer.step() return loss.item()

Step 3: Training loop with loss logging.

device = torch.device("cuda" if torch.cuda.is_available() else "cpu") model = model.to(device) schedule = setup_schedule(T=1000, device=device) losses = [] for epoch in range(5): for batch_idx, (x_0, _) in enumerate(dataloader): x_0 = x_0.to(device) x_0 = x_0 * 2 - 1 # Normalize [0,1] -> [-1,1] loss = train_step(model, x_0, optimizer, schedule) losses.append(loss) if batch_idx % 100 == 0: print(f"Epoch {epoch}, Step {batch_idx}, Loss: {loss:.4f}")

Critical details:

  • Broadcasting: Schedule values indexed by $t$ have shape $(B,)$, but images have shape $(B, C, H, W)$. You must reshape to $(B, 1, 1, 1)$ for correct broadcasting.
  • Normalization: Data must be in $[-1, 1]$. If your DataLoader returns $[0, 1]$, transform with $x \cdot 2 - 1$.
  • Gradient clipping: Clip to norm 1.0 for stability.
  • zero_grad before backward: Forgetting this causes gradient accumulation—loss will explode.

Part 2: Conceptual Questions

Practice explaining these clearly and concisely. For each question, try to answer before revealing the model answer. Imagine you're explaining to a fellow student who has a strong math background but hasn't studied diffusion models.

Core Understanding

1. "Walk me through the forward diffusion process. What happens to an image at each step?"

Show Model Answer

The forward process gradually adds Gaussian noise to a clean image over $T$ timesteps. At each step $t$, the image $x_t$ is a mixture of the original signal and noise:

$$x_t = \sqrt{\bar\alpha_t} \cdot x_0 + \sqrt{1 - \bar\alpha_t} \cdot \epsilon, \quad \epsilon \sim \mathcal{N}(0, I)$$

Concretely: pixel values slowly get erased as noise increases. At $t = 0$, the image is fully intact. At $t = T$, it's indistinguishable from pure Gaussian noise. The key insight is the closed-form shortcut—we can jump from $x_0$ directly to any $x_t$ without iterating through intermediate steps, thanks to the cumulative product $\bar\alpha_t = \prod_{i=1}^{t} \alpha_i$.

Common mistake: Forgetting the signal scaling factor $\sqrt{\bar\alpha_t}$. Without it, the variance of $x_t$ would grow unbounded instead of staying calibrated.

2. "Why do we predict noise instead of predicting the clean image directly?"

Show Model Answer

There are three reasons, building from theory to practice:

  • Mathematical justification: The ELBO derivation shows that maximizing the variational lower bound on log-likelihood reduces to matching the mean of the reverse process Gaussians. The $\epsilon$-reparameterization converts mean-matching into noise-prediction, yielding the simple MSE loss $\|\epsilon - \epsilon_\theta(x_t, t)\|^2$.
  • Uniform difficulty: Predicting $\epsilon$ makes the task roughly equally hard across all timesteps. Predicting $x_0$ directly would be trivial at low noise ($t$ near 0) and impossibly hard at high noise ($t$ near $T$).
  • Simplified loss: Ho et al. showed that dropping the per-timestep weighting from the ELBO (giving uniform weight to all timesteps) actually improves sample quality, and this simplification is natural in the $\epsilon$-parameterization.

Common mistake: Saying "it's just a design choice." There's mathematical justification grounded in the ELBO.

3. "What is the reparameterization trick and why do we need it?"

Show Model Answer

The reparameterization trick separates the randomness from the learned parameters. Instead of sampling $z \sim \mathcal{N}(\mu_\theta, \sigma_\theta^2)$ (which blocks gradient flow because sampling is non-differentiable), we write $z = \mu_\theta + \sigma_\theta \cdot \epsilon$ where $\epsilon \sim \mathcal{N}(0, I)$.

Now gradients can flow through $\mu_\theta$ and $\sigma_\theta$ because $\epsilon$ is a fixed sample—the stochastic node has been moved outside the computational graph. In diffusion models, this is exactly how we express the forward process: $x_t = \sqrt{\bar\alpha_t} \cdot x_0 + \sqrt{1-\bar\alpha_t} \cdot \epsilon$.

Common mistake: Confusing the reparameterization trick (separating randomness for gradient flow) with the noise schedule computation (precomputing $\bar\alpha_t$ values).

4. "Explain the simplified DDPM loss. Why does MSE on noise work?"

Show Model Answer

The chain of reasoning:

  1. ELBO: We can't maximize $\log p(x_0)$ directly, so we maximize a variational lower bound (the ELBO).
  2. KL decomposition: The ELBO decomposes into a sum of KL divergences between consecutive reverse steps.
  3. Gaussians: Because both the forward and reverse processes are Gaussian, each KL reduces to a comparison of means (the variances are fixed).
  4. Mean matching → substitution → $\epsilon$-prediction: The tractable posterior mean $\tilde{\mu}_t$ depends on $x_0$. Rearranging the forward process gives $x_0 = (x_t - \sqrt{1-\bar\alpha_t}\,\epsilon) / \sqrt{\bar\alpha_t}$. Substituting this into $\tilde{\mu}_t$ yields an expression in terms of $\epsilon$. Matching the model's predicted mean to this expression is equivalent to predicting $\epsilon$, giving: $\mathcal{L}_t = \|\epsilon - \epsilon_\theta(x_t, t)\|^2$.
  5. Simplification: The full ELBO has per-timestep weights $\lambda_t$. Ho et al. found that setting $\lambda_t = 1$ for all $t$ (i.e., the unweighted "simple" loss) produces better samples, even though it's no longer a tight bound.

Common mistake: Not connecting the simplified loss back to the ELBO. The simple MSE isn't ad hoc—it's a deliberate simplification of a principled objective.

5. "What's the difference between DDPM and DDIM sampling?"

Show Model Answer

DDPM sampling is stochastic: at each step, we predict the noise, compute the denoised mean, then add fresh noise (controlled by $\beta_t$). Each reverse step is a random draw from a Gaussian, making the full chain a Markov process with $T$ mandatory steps.

DDIM sampling is deterministic (when $\eta = 0$): it rewrites the reverse process as a non-Markovian update that predicts $x_0$ at each step, then jumps to the next noise level. Because the updates aren't constrained to adjacent timesteps, DDIM can skip steps—sampling in 50 steps instead of 1000.

Key facts:

  • Both use the same trained model. DDIM doesn't require retraining.
  • DDIM with $\eta = 0$ is deterministic: same noise → same output, enabling interpolation in latent space.
  • DDIM with $\eta = 1$ recovers DDPM.
  • In practice, DDIM with 50 steps often matches DDPM quality at 1000 steps.

Common mistake: Saying DDIM requires a different model or retraining. It doesn't.

Architecture

6. "Why does the diffusion U-Net use GroupNorm instead of BatchNorm?"

Show Model Answer

BatchNorm computes statistics across the batch dimension. During training with batch size 64, the mean and variance estimates are stable. But during sampling, you typically generate one image at a time (batch size 1), so BatchNorm statistics become degenerate—the "batch mean" is just a single sample.

GroupNorm computes statistics within each sample, dividing channels into groups and normalizing per-group. This makes it completely independent of batch size, so behavior is consistent between training (large batches) and inference (single images).

This is especially important for diffusion models because sampling quality is directly affected by normalization consistency across the iterative denoising chain.

7. "Where do you put attention layers in the U-Net and why?"

Show Model Answer

Self-attention has $O(n^2)$ cost where $n$ is the number of spatial tokens (pixels or patches). For a $32 \times 32$ feature map, that's $1024^2 = 1M$ attention entries—expensive but feasible. For $256 \times 256$, it's $65536^2$—completely impractical.

So attention is placed only at lower resolutions in the U-Net, typically at $16 \times 16$ and $8 \times 8$ feature map sizes. At these resolutions:

  • The computational cost is manageable
  • Each spatial position has a large receptive field, so attention captures global structure (overall composition, symmetry, coherent object shapes)
  • Local details (textures, edges) are handled by convolutions at higher resolutions

Without attention at any resolution, the model can only learn local patterns—leading to images with realistic textures but incoherent global structure.

8. "How does timestep conditioning work?"

Show Model Answer

The model needs to know how much noise is present at each step, so it can adjust its denoising behavior. A single model handles all timesteps $t \in \{0, \ldots, T-1\}$, and timestep conditioning tells it which regime it's operating in.

The pipeline is:

  1. Sinusoidal embedding: Map integer $t$ to a high-dimensional vector using fixed sine/cosine frequencies (same idea as positional encoding in Transformers). This gives the model a smooth, continuous representation of time.
  2. MLP projection: Pass through a small MLP to learn a task-specific representation.
  3. Injection into ResBlocks: The timestep embedding is added to the feature maps inside each residual block (after a linear projection to match dimensions). Some architectures use Adaptive Group Normalization (AdaGN) instead, where $t$ modulates the scale and shift of GroupNorm.

The key insight: passing raw integer $t$ directly would give the model almost no information. The sinusoidal embedding provides a rich, continuous signal that the model can use to distinguish between nearby timesteps.

9. "What are skip connections and why are they important?"

Show Model Answer

Skip connections in the U-Net concatenate encoder feature maps with the corresponding decoder feature maps at the same spatial resolution. They serve two critical roles:

  • Fine spatial detail: The encoder's high-resolution features carry precise spatial information (edges, textures, positions) that would be lost after downsampling to the bottleneck. Skip connections carry this information directly to the decoder.
  • Gradient highway: They provide a short path for gradients to flow from the loss to early layers, preventing vanishing gradients in deep networks.

Without skip connections, the decoder would have to reconstruct all spatial detail from the compressed bottleneck representation alone—resulting in blurry, spatially imprecise outputs. This is particularly devastating for diffusion models, where the model must predict noise at the pixel level.

Guidance & Conditioning

10. "Explain classifier-free guidance. How does training differ?"

Show Model Answer

Training: During training, the class label is randomly replaced with a null token (e.g., NULL_CLASS = num_classes) with some probability, typically 10–20%. This means the same model learns both conditional generation ($p(x | c)$) and unconditional generation ($p(x)$).

Inference: At sampling time, we run two forward passes for each step:

  1. $\epsilon_\text{cond}$: prediction with the real class label
  2. $\epsilon_\text{uncond}$: prediction with the null class label

Then we extrapolate in the direction of the class signal:

$$\epsilon = \epsilon_\text{uncond} + s \cdot (\epsilon_\text{cond} - \epsilon_\text{uncond})$$

When $s = 1$, this is standard conditional generation. When $s > 1$, we amplify the class signal beyond what the model learned—trading diversity for fidelity.

Key advantage over classifier guidance: No separate classifier needed. The guidance signal comes from the diffusion model itself.

Common mistake: Confusing CFG with classifier guidance, which requires training a separate noise-aware classifier and using its gradients to steer sampling.

11. "What does the guidance scale control?"

Show Model Answer

The guidance scale $s$ controls the diversity–fidelity tradeoff:

  • $s = 1$: Standard conditional generation. Natural diversity, but samples may not strongly express the conditioning.
  • $s > 1$ (e.g., 4.0 for MNIST, 7.5 for text-to-image): Amplified conditioning. Sharper, more class-typical samples, but reduced diversity. Acts like an inverse temperature on the conditional distribution.
  • $s < 1$: Weakened conditioning. More diverse but less coherent.
  • Very high $s$ (e.g., 20+): Oversaturated, artifact-prone outputs. The model is being pushed far from its learned distribution.

Intuitively, the guidance scale determines how aggressively the model "steers" toward the condition. Cranking it up makes every "3" look like a textbook "3," but at the cost of all looking the same.

12. "How do negative prompts work mechanically?"

Show Model Answer

In the standard CFG formula, the unconditional prediction $\epsilon_\text{uncond}$ acts as the baseline. Negative prompts replace this baseline:

$$\epsilon = \epsilon_\text{neg} + s \cdot (\epsilon_\text{cond} - \epsilon_\text{neg})$$

Instead of extrapolating away from "no condition," we extrapolate away from the negative condition. This steers the sample away from whatever the negative prompt describes. For example, a negative prompt of "blurry" causes the model to actively move in the opposite direction from "blurry" predictions, resulting in sharper outputs.

Mechanically, the only change is swapping which embedding gets used for the "unconditional" forward pass. The rest of the pipeline is identical.

Advanced

13. "What's the difference between pixel-space and latent diffusion?"

Show Model Answer

Pixel-space diffusion operates directly on raw pixels. For a $256 \times 256 \times 3$ image, the diffusion process runs on a $256 \times 256 \times 3$ tensor at every step. This is conceptually simple but computationally expensive—especially because attention is $O(n^2)$ in spatial dimensions.

Latent diffusion (Rombach et al., 2022) first trains a VAE to compress images into a smaller latent space (e.g., $256 \times 256 \times 3 \to 32 \times 32 \times 4$, an $8\times$ spatial reduction), then runs the entire diffusion process in that compressed space. At generation time, the final latent is decoded back to pixel space by the VAE decoder.

Tradeoffs:

  • Speed: Latent diffusion is ~10× faster because the spatial dimensions are 8× smaller in each direction ($64\times$ fewer spatial tokens).
  • Quality: Slight quality loss from VAE compression, but the diffusion model can focus on semantic content rather than imperceptible pixel-level detail.
  • Complexity: Requires training a separate VAE first, adding a two-stage pipeline.

Stable Diffusion is the canonical example of latent diffusion in practice.

14. "What are the tradeoffs between $\epsilon$-prediction, $x_0$-prediction, and $v$-prediction?"

Show Model Answer

All three parameterizations are mathematically equivalent—you can convert between them given $x_t$ and $\bar\alpha_t$. The difference is in training dynamics:

  • $\epsilon$-prediction: The model predicts the noise. Stable at high noise levels (large $t$) because the target $\epsilon$ has constant variance regardless of $t$. But at low noise (small $t$), the signal-to-noise ratio is high, and predicting noise becomes an afterthought.
  • $x_0$-prediction: The model predicts the clean image directly. Natural at low noise, but unstable at high noise because the model must hallucinate an entire image from near-pure noise.
  • $v$-prediction: The model predicts $v = \sqrt{\bar\alpha_t} \cdot \epsilon - \sqrt{1-\bar\alpha_t} \cdot x_0$, a linear combination that balances the difficulty across all timesteps. This gives more uniform gradient magnitudes and is especially useful with cosine schedules.

In practice, $\epsilon$-prediction is the most common (DDPM, Stable Diffusion v1). $v$-prediction is used in newer models (Stable Diffusion v2, Imagen Video) for better training stability.

15. "What is flow matching and how does it differ from diffusion?"

Show Model Answer

Flow matching learns a velocity field that transports samples from noise to data along straight (or near-straight) paths. Instead of the diffusion framework of gradual corruption and iterative denoising, flow matching defines a continuous trajectory:

$$x_t = (1 - t) \cdot x_0 + t \cdot x_1, \quad t \in [0, 1]$$

where $x_0$ is a data sample and $x_1 \sim \mathcal{N}(0, I)$ is a noise sample. At $t = 0$ we have pure data; at $t = 1$ we have pure noise. The model learns the velocity $v_\theta(x_t, t)$ that moves $x_t$ along this path. The training objective is simply:

$$\mathcal{L} = \|v_\theta(x_t, t) - (x_1 - x_0)\|^2$$

Key differences from diffusion:

  • Simpler objective: No noise schedule, no $\bar\alpha_t$ bookkeeping, no ELBO derivation.
  • Straighter paths: The interpolation between noise and data is (approximately) linear, which means fewer ODE steps are needed at inference.
  • Fewer steps: Straight paths need fewer ODE solver steps. With rectified flows (iteratively straightening trajectories), models like Stable Diffusion 3 can generate high-quality images in as few as 4–8 steps.
  • Same model architectures: U-Nets and DiTs work with flow matching; only the training target changes.

16. "How would you scale this model to generate 512×512 images?"

Show Model Answer

This is a system design question—see Part 4 for the full discussion. In brief:

  • Latent diffusion: Train a VAE to compress $512 \to 64$ spatially, then run diffusion in latent space.
  • Architecture: U-Net with selective attention ($32 \times 32$ and below), or DiT (flat transformer over all patches). DiT scales more predictably with compute.
  • Training: Mixed precision (bf16), gradient checkpointing, distributed training, EMA weights.
  • Inference: DDIM with 50 steps, classifier-free guidance, dynamic thresholding.

Show All 10 Bugs and Fixes

Bug 1: Wrong scaling in forward process.

The q_sample function uses alphas[t] ($\alpha_t$) instead of alpha_bar[t] ($\bar\alpha_t$). The forward process requires the cumulative product. Using per-step $\alpha_t$ means the noising is wrong at every timestep—the noise levels don't correspond to what the model expects.

# Bug: alpha_t = alphas[t] # Fix: alpha_t = alpha_bar[t]

Bug 2: Missing sqrt in forward process.

The forward process computes alpha_t * x_0 + (1 - alpha_t) * noise instead of sqrt(alpha_bar_t) * x_0 + sqrt(1 - alpha_bar_t) * noise. Without the square roots, the means and variances of $x_t$ are wrong.

# Bug: alpha_t * x_0 + (1 - alpha_t) * noise # Fix: sqrt(alpha_bar_t) * x_0 + sqrt(1 - alpha_bar_t) * noise

Bug 3: Off-by-one in sampling loop.

The condition if t_idx >= 0 adds noise at every step, including the final step $t=0$. The last step should be noiseless—the mean at $t=0$ is our final clean output. Adding noise at $t=0$ corrupts the result.

# Bug: if t_idx >= 0: x = mean + sqrt(beta) * noise # Fix: if t_idx > 0: x = mean + sqrt(beta) * noise # else: x = mean

Bug 4: Sampling with training weights instead of EMA.

The code carefully maintains an EMA model during training but then calls sample(model) instead of sample(ema_model). The entire point of EMA (from Lecture 6) is to use the smoothed weights for generation—they produce cleaner, higher-quality samples.

# Bug: images = sample(model) # Fix: images = sample(ema_model)

Bug 5: Missing optimizer.zero_grad().

The training loop calls loss.backward() and optimizer.step() but never calls optimizer.zero_grad(). Gradients accumulate across batches, causing the effective gradient to explode. Training loss will diverge rapidly.

# Bug: loss.backward(); optimizer.step() # Fix: optimizer.zero_grad(); loss.backward(); optimizer.step()

Bug 6: Wrong data normalization.

The transform only applies ToTensor(), which maps pixel values to $[0, 1]$. But the diffusion model expects data in $[-1, 1]$. During sampling, the reverse process produces values centered around 0, but the model was trained on data centered around 0.5. This causes a systematic color shift in all outputs.

# Bug: transforms.Compose([transforms.ToTensor()]) # Fix: transforms.Compose([transforms.ToTensor(), # transforms.Normalize([0.5], [0.5])]) # Or: x_0 = x_0 * 2 - 1 in the training loop

Bug 7: Timestep embedding passes raw integers.

The TimestepEmbedding just broadcasts the raw integer $t$ to fill a vector. This gives the model almost no ability to distinguish between timesteps—the values 499 and 500 look nearly identical. It should use sinusoidal positional embedding to create a rich, continuous representation.

# Bug: return t.float().unsqueeze(-1).expand(-1, self.dim) # Fix: Use sinusoidal embedding: # freqs = 1 / (10000 ** (arange(half_dim) / half_dim)) # emb = cat([sin(t * freqs), cos(t * freqs)], dim=-1)

Bug 8: CFG formula is wrong.

The sampling code uses eps_uncond + scale * eps_cond instead of the correct extrapolation formula eps_uncond + scale * (eps_cond - eps_uncond). Without the subtraction, there's no extrapolation—guidance doesn't amplify the class signal.

# Bug: eps = eps_uncond + guidance_scale * eps_cond # Fix: eps = eps_uncond + guidance_scale * (eps_cond - eps_uncond)

Bug 9: Device mismatch.

The schedule tensors (betas, alphas, alpha_bar) are created on CPU, but the model and data are on GPU/MPS. When the sampling function indexes alphas[t_idx] and uses it with x (on device), this causes a device mismatch error or, worse, silent incorrect computation.

# Bug: betas = torch.linspace(1e-4, 0.06, T) # CPU # Fix: betas = torch.linspace(1e-4, 0.06, T).to(DEVICE) # alphas = ... .to(DEVICE) # alpha_bar = ... .to(DEVICE)

Bug 10: Beta schedule upper bound too high.

The beta schedule goes up to $0.06$, which is 3$\times$ higher than the standard DDPM value of $0.02$ (Ho et al. 2020). At $\beta_t = 0.06$, $\bar\alpha_T$ reaches near-zero far too quickly, meaning the model destroys too much signal too fast. This causes numerical instability and poor sample quality. The fix is to use the standard upper bound:

# Bug: betas = torch.linspace(1e-4, 0.06, T) # Fix: betas = torch.linspace(1e-4, 0.02, T)

Part 4: System Design Discussion

The Prompt
"You've built a diffusion model that works on 28×28 MNIST. Your manager asks you to scale it to generate 512×512 photorealistic images. Walk through your approach."

This is an open-ended discussion. Think through the problem carefully before reading the reference answer. Consider architecture, training, inference, and evaluation.

Show Reference Answer

The Resolution Problem

Going from $28 \times 28$ to $512 \times 512$ is a $335\times$ increase in pixel count. You cannot simply make the U-Net bigger—self-attention is $O(n^2)$ in the number of spatial tokens, so a $512 \times 512$ attention layer would be $65536^2 \approx 4 \times 10^9$ entries. You need a fundamentally different approach.

Architecture Decisions

  • Latent diffusion: Train a VAE to compress images spatially ($512 \times 512 \times 3 \to 64 \times 64 \times 4$, an $8\times$ reduction in each spatial dimension). Run the entire diffusion process in the $64 \times 64 \times 4$ latent space. Decode back to pixels only at the final step. This reduces compute by $\sim64\times$.
  • DiT vs U-Net: Two architecture families to choose from. U-Net: encoder-decoder with skip connections and selective attention placement (only at $32 \times 32$ and below, since attention is $O(n^2)$ in spatial tokens). DiT: patchify the latent into a flat token sequence, process with repeated transformer blocks (uniform attention over all patches), unpatchify back. DiT is architecturally simpler (no multi-resolution hierarchy) and scales more predictably with compute.
  • Efficient attention: Regardless of architecture, use flash attention or memory-efficient variants to reduce the constant factor in attention cost.
  • Gradient checkpointing: Trade compute for memory by recomputing intermediate activations during the backward pass instead of storing them. Essential for training large models on limited GPU memory.
  • Mixed precision: Train in fp16 or bf16 for approximately $2\times$ memory savings and faster matmuls.

Training Decisions

  • Dataset: MNIST's 60K grayscale digits are nowhere near sufficient. You need 100K+ diverse, high-resolution images at minimum. For photorealistic generation, datasets like LAION-5B (billions of image-text pairs) are used in practice.
  • Training time: Expect 100K–1M training steps, taking days to weeks on a cluster of 8+ GPUs.
  • Batch size: 256+ with distributed data-parallel training across multiple GPUs.
  • EMA: Critical at this scale. Use a high decay rate like $0.9999$ (more smoothing than the $0.999$ used for MNIST).
  • Schedule: Cosine schedule with zero terminal SNR for better coverage of the noise range.

Inference Decisions

  • Sampler: DDIM with 50 steps, not 1000-step DDPM. For production, consider DPM-Solver++ or other high-order solvers that achieve good quality in 20–25 steps.
  • CFG: Guidance scale around $7.5$ for text-to-image (higher than the $\sim4.0$ used for class-conditional MNIST).
  • Dynamic thresholding: At high guidance scales, predicted $x_0$ values can exceed $[-1, 1]$, causing saturation artifacts. Dynamic thresholding (Imagen) clips to a percentile of the predicted values instead of a fixed range.
  • Batched generation: Generate multiple images in parallel to maximize GPU utilization.

Follow-up Questions to Consider

  • "What's the memory bottleneck?" Attention layers at the lowest resolution, plus storing activations for backprop. Solutions: gradient checkpointing, flash attention, mixed precision.
  • "How would you add text conditioning?" Replace class embeddings with CLIP text embeddings. Cross-attention layers in the U-Net/DiT attend to the text token sequence. Train with random caption dropout for CFG.
  • "What metrics would you use?" FID (Frechet Inception Distance) for distribution-level quality, CLIP score for text-image alignment, human evaluation for subjective quality.
  • "How would you reduce inference latency?" Distillation (progressive or consistency distillation) to reduce from 50 steps to 1–4 steps. Model quantization (INT8). Speculative decoding. Caching attention KV across steps when possible.

General Tips

Approach

These challenges help you identify gaps in your understanding. Approach them as self-evaluation, not performance. Getting stuck is information—it tells you exactly what to study next.

  1. Think out loud. Narrate your reasoning as you code—it helps you catch your own mistakes and clarifies your mental model.
